### 3.1 状态编码函数

```
FUNCTION StateEncode(raw_state) -> encoded_state:
    """
    将11维原始状态编码为神经网络输入
    输入: raw_state = [x, y, z, vx, vy, vz, d_goal, d_obs, d_uav, E_remain, p_delivery]
    输出: encoded_state ∈ R^(10 + M + 1)
    """

    // ---------- 参数初始化 ----------
    s_min = [0, 0, 10, -15, -15, -15, 0, 0, 0, 0]
    s_max = [100, 100, 120, 15, 15, 15, 180, 180, 180, 1]
    M = MAX_ORDERS                  // 总订单数 (如20)

    // ---------- 分离连续与离散状态 ----------
    continuous_state = raw_state[0:10]      // 前10维: 连续状态
    discrete_state = INT(raw_state[10])     // 第11维: 配送进度

    // ---------- 连续状态归一化 (Min-Max) ----------
    normalized = ZEROS(10)
    FOR i = 0 TO 9:
        normalized[i] = (continuous_state[i] - s_min[i]) / (s_max[i] - s_min[i] + 1e-8)
        normalized[i] = CLIP(normalized[i], 0.0, 1.0)
    END FOR

    // ---------- 离散状态独热编码 ----------
    one_hot = ZEROS(M + 1)
    one_hot[discrete_state] = 1.0

    // ---------- 状态拼接 ----------
    encoded_state = CONCATENATE(normalized, one_hot)

    RETURN encoded_state
    // encoded_state ∈ R^(10 + M + 1), 对于M=20, 维度为31
END FUNCTION


FUNCTION EncodeGlobalState(all_observations, shared_info) -> global_state:
    """
    将所有无人机的局部观测和共享信息编码为全局状态
    输入: all_observations ∈ R^(N x 11), shared_info ∈ R^D_shared
    输出: global_state ∈ R^(N*11 + D_shared)
    """

    N = all_observations.rows          // 无人机数量
    D_obs = all_observations.cols      // 局部观测维度 = 11

    // 拼接所有无人机的局部观测
    flat_obs = FLATTEN(all_observations)  // shape: (N * D_obs,)

    // 拼接共享全局信息 (目标位置、障碍物信息等)
    global_state = CONCATENATE(flat_obs, shared_info)

    RETURN global_state
    // 对于N=3, D_shared=10: global_state ∈ R^43
END FUNCTION
```

### 3.2 动作选择函数

```
FUNCTION SelectAction(actor_network, observation, deterministic=FALSE) -> action, log_prob:
    """
    使用Actor网络选择动作
    输入: actor_network (网络参数θ), observation ∈ R^D_obs, 是否确定性模式
    输出: action ∈ R^3 (三维加速度), log_prob ∈ R
    """

    // ---------- 前向传播 ----------
    // 隐藏层计算
    h1 = ReLU(W1 * observation + b1)      // shape: (256,)
    h2 = ReLU(W2 * h1 + b2)               // shape: (128,)
    h3 = ReLU(W3 * h2 + b3)               // shape: (64,)

    // 输出层
    mu = W_mu * h3 + b_mu                  // 动作均值, shape: (3,)
    log_sigma = W_sigma * h3 + b_sigma     // 对数标准差, shape: (3,)
    log_sigma = CLIP(log_sigma, -20, 2)    // 数值稳定性
    sigma = EXP(log_sigma)

    // ---------- 动作采样 ----------
    IF deterministic THEN:
        raw_action = mu
    ELSE:
        epsilon ~ N(0, I)                  // 标准正态采样
        raw_action = mu + sigma * epsilon
    END IF

    // ---------- tanh压缩 ----------
    tanh_action = TANH(raw_action)         // ∈ [-1, 1]^3

    // ---------- 线性映射到加速度空间 ----------
    a_raw = [5.0 * tanh_action[0],
             5.0 * tanh_action[1],
             3.0 * tanh_action[2]]         // ∈ [-5,5] x [-5,5] x [-3,3]

    // ---------- 对数概率计算 (含Jacobian修正) ----------
    log_prob_gauss = -0.5 * SUM(
        ((raw_action - mu)^2 / (sigma^2)) + LOG(2 * PI * sigma^2)
    )
    log_prob_jacobian = SUM(LOG(1 - tanh_action^2 + 1e-6))
    log_prob = log_prob_gauss - log_prob_jacobian

    RETURN a_raw, log_prob
END FUNCTION
```

### 3.3 动作施加与状态更新函数

```
FUNCTION ApplyAction(action_raw, current_state, prev_action, prev_smooth) -> new_state, action_final:
    """
    施加动作并更新状态 (含四层约束处理)
    输入: action_raw ∈ R^3, current_state, prev_action ∈ R^3, prev_smooth ∈ R^3
    输出: new_state, action_final ∈ R^3
    """

    // ========== 层1: 加速度边界裁剪 ==========
    a_clipped[0] = CLIP(action_raw[0], -AX_MAX, AX_MAX)   // [-5, 5]
    a_clipped[1] = CLIP(action_raw[1], -AY_MAX, AY_MAX)   // [-5, 5]
    a_clipped[2] = CLIP(action_raw[2], -AZ_MAX, AZ_MAX)   // [-3, 3]

    // ========== 层1: Jerk限制 ==========
    delta_a = a_clipped - prev_action
    delta_a_limited[0] = CLIP(delta_a[0], -JERK_MAX_XY * DT, JERK_MAX_XY * DT)
    delta_a_limited[1] = CLIP(delta_a[1], -JERK_MAX_XY * DT, JERK_MAX_XY * DT)
    delta_a_limited[2] = CLIP(delta_a[2], -JERK_MAX_Z * DT, JERK_MAX_Z * DT)
    a_jerk_limited = prev_action + delta_a_limited

    // ========== EMA低通滤波 (自适应alpha) ==========
    d_obs = current_state.d_obs
    alpha = ALPHA_BASE + (1 - ALPHA_BASE) * MIN(d_obs / D_SAFE_UAV, 1.0)
    a_smooth = alpha * a_jerk_limited + (1 - alpha) * prev_smooth

    // ========== 层2: 速度约束 (半隐式欧拉积分) ==========
    // 先更新速度
    v_new = current_state.velocity + a_smooth * DT
    speed = NORM(v_new)
    IF speed > V_MAX THEN:
        v_new = v_new * (V_MAX / speed)
        a_smooth = (v_new - current_state.velocity) / DT  // 回推等效加速度
    END IF

    // ========== 层1: 位置边界约束 (使用更新后的速度) ==========
    // 后更新位置
    p_new = current_state.position + v_new * DT

    // X轴边界
    IF p_new.x > X_RANGE[1] - BOUNDARY_WARN AND v_new.x > 0 THEN:
        a_smooth[0] = MIN(a_smooth[0], -K_BOUNDARY * v_new.x)
    ELSE IF p_new.x < X_RANGE[0] + BOUNDARY_WARN AND v_new.x < 0 THEN:
        a_smooth[0] = MAX(a_smooth[0], -K_BOUNDARY * v_new.x)
    END IF
    p_new.x = CLIP(p_new.x, X_RANGE[0], X_RANGE[1])

    // Y轴边界 (同理)
    IF p_new.y > Y_RANGE[1] - BOUNDARY_WARN AND v_new.y > 0 THEN:
        a_smooth[1] = MIN(a_smooth[1], -K_BOUNDARY * v_new.y)
    ELSE IF p_new.y < Y_RANGE[0] + BOUNDARY_WARN AND v_new.y < 0 THEN:
        a_smooth[1] = MAX(a_smooth[1], -K_BOUNDARY * v_new.y)
    END IF
    p_new.y = CLIP(p_new.y, Y_RANGE[0], Y_RANGE[1])

    // Z轴高度约束
    IF p_new.z < Z_RANGE[0] THEN:
        p_new.z = Z_RANGE[0]
        v_new.z = MAX(0, v_new.z)
    END IF
    IF p_new.z > Z_RANGE[1] THEN:
        p_new.z = Z_RANGE[1]
        v_new.z = MIN(0, v_new.z)
    END IF

    // 构建新状态
    new_state = current_state
    new_state.position = p_new
    new_state.velocity = v_new

    RETURN new_state, a_smooth
END FUNCTION
```

### 3.4 碰撞检测函数

```
FUNCTION CheckCollision(positions, obstacles, obstacle_radii, boundary) -> collision_detected, details:
    """
    检测所有类型的碰撞
    输入: positions ∈ R^(N,3), obstacles, obstacle_radii, boundary
    输出: collision_detected (布尔), details (字典)
    """

    N = LEN(positions)
    details = {"uav_collision": FALSE, "obs_collision": FALSE, "boundary_violation": FALSE}

    // ---------- 无人机间碰撞检测 ----------
    FOR i = 0 TO N-1:
        FOR j = i+1 TO N-1:
            dist = NORM(positions[i] - positions[j])
            IF dist < 2 * COLLISION_RADIUS THEN:
                details["uav_collision"] = TRUE
                RETURN TRUE, details
            END IF
        END FOR
    END FOR

    // ---------- 障碍物碰撞检测 ----------
    FOR i = 0 TO N-1:
        FOR k = 0 TO LEN(obstacles)-1:
            dist = NORM(positions[i] - obstacles[k])
            IF dist < (COLLISION_RADIUS + obstacle_radii[k]) THEN:
                details["obs_collision"] = TRUE
                RETURN TRUE, details
            END IF
        END FOR
    END FOR

    // ---------- 越界碰撞检测 ----------
    FOR i = 0 TO N-1:
        FOR dim = 0 TO 2:
            IF positions[i][dim] < boundary[dim][0] OR positions[i][dim] > boundary[dim][1]:
                details["boundary_violation"] = TRUE
                RETURN TRUE, details
            END IF
        END FOR
    END FOR

    RETURN FALSE, details
END FUNCTION
```

### 3.5 奖励计算函数

```
FUNCTION CalculateReward(
    positions,           // (N,3) 当前位置
    prev_positions,      // (N,3) 上一步位置
    actions,             // (N,3) 当前动作
    prev_actions,        // (N,3) 上一步动作
    velocities,          // (N,3) 当前速度
    goal_pos,            // (3,) 主目标位置
    goal_positions,      // (K,3) 各配送目标位置
    formation_offsets,   // (N,3) 期望队形偏移
    obstacles,           // (M,4) 障碍物 [x,y,z,radius]
    cargo_loaded,        // (K,) 货物是否已装载
    cargo_delivered,     // (K,) 货物是否已送达
    battery_ratios,      // (N,) 各无人机剩余电量比
    step_count,          // 当前步数
    max_steps            // 最大步数
) -> total_reward, reward_components:
    """
    计算8分量加权奖励 + 势能塑形
    输出: total_reward (标量), reward_components (字典)
    """

    N = LEN(positions)
    K = LEN(goal_positions)
    rewards = {}

    // ========== 1. 目标到达奖励 R_goal ==========
    dist_to_goal = NORM(MEAN(positions, axis=0) - goal_pos)
    IF dist_to_goal <= GOAL_THRESHOLD THEN:
        rewards["R_goal"] = R_GOAL          // +200
    ELSE:
        rewards["R_goal"] = 0.0
    END IF

    // ========== 2. 距离引导奖励 R_dist ==========
    IF prev_positions != NULL THEN:
        prev_dist = NORM(MEAN(prev_positions, axis=0) - goal_pos)
        rewards["R_dist"] = C_DIST * (prev_dist - dist_to_goal)
    ELSE:
        rewards["R_dist"] = 0.0
    END IF

    // ========== 3. 碰撞惩罚 P_collision ==========
    collision_detected, _ = CheckCollision(positions, obstacles, obstacle_radii, boundary)
    IF collision_detected THEN:
        rewards["P_collision"] = P_COLLISION  // -200
    ELSE:
        rewards["P_collision"] = 0.0
    END IF

    // ========== 4. 安全距离惩罚 P_safe ==========
    penalty_safe = 0.0
    // 无人机间预警
    FOR i = 0 TO N-1:
        FOR j = i+1 TO N-1:
            dist = NORM(positions[i] - positions[j])
            IF dist < D_SAFE_UAV THEN:
                ratio = MAX(0, (D_SAFE_UAV - dist) / (D_SAFE_UAV - 2*COLLISION_RADIUS))
                penalty_safe -= 10.0 * ratio^2
            END IF
        END FOR
    END FOR
    rewards["P_safe"] = penalty_safe

    // ========== 5. 间距惩罚 P_proximity ==========
    penalty_prox = 0.0
    FOR i = 0 TO N-1:
        FOR j = i+1 TO N-1:
            dist = NORM(positions[i] - positions[j])
            penalty_prox += MAX(0, 1.0 - dist / D_SAFE_UAV)
        END FOR
    END FOR
    rewards["P_proximity"] = P_PROXIMITY_MAX * penalty_prox

    // ========== 6. 队形保持惩罚 P_formation ==========
    IF N > 1 THEN:
        center = MEAN(positions, axis=0)
        desired_pos = center + formation_offsets[0:N]
        formation_errors = [NORM(positions[i] - desired_pos[i]) FOR i = 0 TO N-1]
        rewards["P_formation"] = -LAMBDA_FORM * MEAN(formation_errors)
    ELSE:
        rewards["P_formation"] = 0.0
    END IF

    // ========== 7. 能耗惩罚 P_energy ==========
    energy_consumptions = []
    FOR i = 0 TO N-1:
        e_accel = NORM(actions[i])^2                   // 加速度能耗
        e_velocity = 0.1 * NORM(velocities[i])^2       // 速度维持能耗
        IF prev_positions != NULL THEN:
            e_altitude = 2.0 * ABS(positions[i][2] - prev_positions[i][2])
        ELSE:
            e_altitude = 0.0
        END IF
        energy_consumptions.APPEND(e_accel + e_velocity + e_altitude)
    END FOR
    avg_battery = MEAN(battery_ratios)
    adaptive_lambda = LAMBDA_ENERGY * (1.0 + 2.0 * (1.0 - avg_battery))
    rewards["P_energy"] = -adaptive_lambda * MEAN(energy_consumptions)

    // ========== 8. 时间惩罚 P_time ==========
    rewards["P_time"] = -LAMBDA_TIME                   // -2.0

    // ========== 9. 平滑惩罚 P_smooth ==========
    IF prev_actions != NULL AND step_count > 0 THEN:
        action_diff = actions - prev_actions
        rewards["P_smooth"] = -LAMBDA_SMOOTH * MEAN(NORM_EACH(action_diff)^2)
    ELSE:
        rewards["P_smooth"] = 0.0
    END IF

    // ========== 10. 配送完成奖励 R_delivery ==========
    R_delivery = 0.0
    newly_delivered = []
    FOR k = 0 TO K-1:
        dist = NORM(MEAN(positions, axis=0) - goal_positions[k])
        IF dist <= DROP_THRESHOLD AND cargo_loaded[k] AND NOT cargo_delivered[k] THEN:
            delivered_count = SUM(cargo_delivered)
            r_k = R_DELIVERY * (1.0 + 0.5 * delivered_count / MAX(K, 1))
            R_delivery += r_k
            newly_delivered.APPEND(k)
        END IF
    END FOR
    rewards["R_delivery"] = R_delivery

    // ========== 11. 全部完成奖励 R_all_done ==========
    IF ALL(cargo_delivered) THEN:
        rewards["R_all_done"] = R_ALL_DONE             // +300
    ELSE:
        rewards["R_all_done"] = 0.0
    END IF

    // ========== 12. 超时惩罚 P_timeout ==========
    IF step_count >= max_steps THEN:
        rewards["P_timeout"] = P_TIMEOUT               // -100
    ELSE:
        rewards["P_timeout"] = 0.0
    END IF

    // ========== 13. 势能塑形奖励 ==========
    IF prev_positions != NULL THEN:
        remaining_targets = [goal_positions[k] FOR k WHERE NOT cargo_delivered[k]]
        IF LEN(remaining_targets) > 0 THEN:
            prev_min_dist = MIN([NORM(MEAN(prev_positions) - t) FOR t IN remaining_targets])
            curr_min_dist = MIN([NORM(MEAN(positions) - t) FOR t IN remaining_targets])
            shaping = 0.5 * (GAMMA * (-curr_min_dist) - (-prev_min_dist))
        ELSE:
            shaping = 0.0
        END IF
    ELSE:
        shaping = 0.0
    END IF
    rewards["shaping"] = shaping

    // ========== 计算总奖励 ==========
    total_reward = SUM(rewards.values())

    RETURN total_reward, rewards
END FUNCTION
```

### 3.6 任务分配函数

#### 3.6.1 代价矩阵构建

```
FUNCTION BuildCostMatrix(drones, orders, current_time) -> cost_matrix:
    """
    构建 N x M 代价矩阵
    输入: drones (N架), orders (M个), current_time
    输出: cost_matrix ∈ R^(N x M)
    """

    N = LEN(drones)
    M = LEN(orders)
    cost_matrix = ZEROS(N, M)

    FOR i = 0 TO N-1:
        FOR j = 0 TO M-1:
            drone = drones[i]
            order = orders[j]

            // 距离代价: 当前位置 -> 取货点 -> 配送点
            dist_pickup = DISTANCE(drone.position, order.pickup_pos)
            dist_delivery = DISTANCE(order.pickup_pos, order.delivery_pos)
            total_distance = dist_pickup + dist_delivery

            // 时间代价
            estimated_time = total_distance / drone.avg_speed + 30.0  // +悬停时间

            // 能耗代价 (单位距离能耗 0.5 Wh/m)
            energy_consumption = total_distance * 0.5

            // 紧急度代价
            time_remaining = order.deadline - current_time
            IF time_remaining > 0 THEN:
                urgency = estimated_time / time_remaining
            ELSE:
                urgency = 10.0
            END IF

            // 负载均衡因子
            load_factor = LEN(drone.task_list) / 10.0
                + (drone.max_energy - drone.remaining_energy) / drone.max_energy

            // 综合代价
            cost_matrix[i][j] = ALPHA_COST * total_distance
                               + BETA_COST * estimated_time
                               + GAMMA_COST * energy_consumption
                               + DELTA_COST * urgency
                               + LAMBDA_BALANCE * load_factor
        END FOR
    END FOR

    RETURN cost_matrix
END FUNCTION
```

#### 3.6.2 约束检查

```
FUNCTION CheckConstraints(drone, order, current_time) -> is_valid, reason:
    """
    检查分配约束: 容量、电量、时间窗口
    """

    // 容量约束
    IF drone.current_load + order.weight > drone.max_load THEN:
        RETURN FALSE, "容量超限"
    END IF

    // 电量约束
    total_distance = DISTANCE(drone.position, order.pickup_pos)
                   + DISTANCE(order.pickup_pos, order.delivery_pos)
    energy_required = total_distance * 0.5                    // Wh
    IF drone.remaining_energy < energy_required * (1 + ENERGY_SAFE_MARGIN) THEN:
        RETURN FALSE, "电量不足"
    END IF

    // 时间窗口约束
    flight_time = total_distance / drone.avg_speed
    estimated_arrival = current_time + flight_time + 30.0     // +悬停时间
    IF estimated_arrival > order.deadline * (1 - TIME_SAFE_MARGIN) THEN:
        RETURN FALSE, "无法按时送达"
    END IF

    RETURN TRUE, "满足所有约束"
END FUNCTION
```

#### 3.6.3 贪心分配算法

```
FUNCTION GreedyAllocate(orders, drones, current_time) -> allocation:
    """
    贪心任务分配算法
    时间复杂度: O(M^2)
    输入: 订单列表, 无人机列表, 当前时间
    输出: allocation = {drone_id: [order_id, ...]}
    """

    allocation = {drone.id: [] FOR drone IN drones}
    unassigned = COPY(orders)
    cost_matrix = BuildCostMatrix(drones, orders, current_time)

    WHILE LEN(unassigned) > 0:
        assigned_this_round = FALSE

        FOR each drone IN drones:
            IF NOT drone.is_available OR LEN(unassigned) == 0:
                CONTINUE
            END IF

            best_order = NULL
            best_cost = INF
            best_idx = -1

            // 找到代价最小的可行订单
            FOR j = 0 TO LEN(unassigned)-1:
                order = unassigned[j]
                original_idx = orders.INDEX_OF(order)
                cost = cost_matrix[drone.id_idx][original_idx]

                is_valid, _ = CheckConstraints(drone, order, current_time)
                IF is_valid AND cost < best_cost THEN:
                    best_cost = cost
                    best_order = order
                    best_idx = j
                END IF
            END FOR

            // 执行分配
            IF best_order != NULL THEN:
                allocation[drone.id].APPEND(best_order.id)
                drone.task_list.APPEND(best_order.id)
                drone.current_load += best_order.weight
                best_order.status = ASSIGNED
                unassigned.REMOVE(best_idx)
                assigned_this_round = TRUE
            END IF
        END FOR

        IF NOT assigned_this_round THEN:
            BREAK                   // 无法继续分配
        END IF
    END WHILE

    RETURN allocation
END FUNCTION
```

#### 3.6.4 匈牙利分配算法

```
FUNCTION HungarianAllocate(orders, drones, current_time) -> allocation:
    """
    匈牙利任务分配算法
    时间复杂度: O(max(N,M)^3)
    保证全局最优 (单目标)
    """

    N = LEN(drones)
    M = LEN(orders)
    IF N == 0 OR M == 0 THEN:
        RETURN {}
    END IF

    // 构建代价矩阵
    cost_matrix = BuildCostMatrix(drones, orders, current_time)

    // 扩展为方阵
    n = MAX(N, M)
    extended_cost = ZEROS(n, n)
    IF N < M THEN:
        extended_cost[0:N, 0:M] = cost_matrix
        extended_cost[N:n, :] = 1e6        // 虚拟无人机高代价
    ELSE IF N > M THEN:
        extended_cost[0:N, 0:M] = cost_matrix
        // 虚拟订单代价为0
    ELSE:
        extended_cost = cost_matrix
    END IF

    // 调用匈牙利算法 (行规约 -> 列规约 -> 寻找独立零元素 -> 迭代调整)
    row_ind, col_ind = HungarianAlgorithm(extended_cost)

    // 构建分配结果
    allocation = {drone.id: [] FOR drone IN drones}
    FOR each (i, j) IN (row_ind, col_ind):
        IF i >= N OR j >= M THEN:
            CONTINUE            // 跳过虚拟分配
        END IF

        drone = drones[i]
        order = orders[j]

        is_valid, _ = CheckConstraints(drone, order, current_time)
        IF is_valid THEN:
            allocation[drone.id].APPEND(order.id)
            drone.task_list.APPEND(order.id)
            drone.current_load += order.weight
            order.status = ASSIGNED
        END IF
    END FOR

    RETURN allocation
END FUNCTION
```

#### 3.6.5 自适应分配器

```
FUNCTION AdaptiveAllocate(orders, drones, current_time, is_static=TRUE) -> allocation:
    """
    根据场景自动选择分配算法
    """

    num_orders = LEN(orders)
    urgency = ComputeUrgency(orders, current_time)

    IF num_orders <= 3 OR urgency > 0.8 THEN:
        // 订单少或紧急度高: 贪心算法 (实时性优先)
        allocation = GreedyAllocate(orders, drones, current_time)
    ELSE IF num_orders <= 20 AND is_static THEN:
        // 订单适中且静态: 匈牙利算法 (最优性优先)
        allocation = HungarianAllocate(orders, drones, current_time)
    ELSE:
        // 复杂场景: 匈牙利初始分配
        allocation = HungarianAllocate(orders, drones, current_time)
    END IF

    RETURN allocation
END FUNCTION


FUNCTION ComputeUrgency(orders, current_time) -> urgency:
    """
    计算整体订单紧急度
    """
    IF LEN(orders) == 0 THEN:
        RETURN 0.0
    END IF

    scores = []
    FOR each order IN orders:
        time_ratio = (order.deadline - current_time)
                   / MAX(order.deadline - order.created_time, 1)
        scores.APPEND(1 - time_ratio)
    END FOR

    RETURN MEAN(scores)
END FUNCTION
```

#### 3.6.6 动态重分配

```
FUNCTION HandleDynamicEvent(event, drones, orders, current_allocation, current_time) -> updated_allocation:
    """
    处理动态事件并执行重分配
    """

    CASE event.type OF:

    // ----- 新订单到达 -----
    NEW_ORDER:
        pending_orders = FILTER(orders, status == PENDING)
        idle_drones = FILTER(drones, is_available)
        IF LEN(idle_drones) > 0 AND LEN(pending_orders) > 0 THEN:
            new_alloc = GreedyAllocate(pending_orders, idle_drones, current_time)
            updated_allocation = MERGE(current_allocation, new_alloc)
        ELSE:
            updated_allocation = current_allocation
        END IF

    // ----- 电量预警 -----
    BATTERY_LOW:
        low_batt_drone = FIND(drones, id == event.drone_id)
        low_batt_drone.status = CHARGING

        // 找出该无人机未取货的订单
        orders_to_reassign = FILTER(
            orders WHERE id IN current_allocation[low_batt_drone.id]
                      AND status == ASSIGNED
        )
        FOR each order IN orders_to_reassign:
            order.status = PENDING
        END FOR

        IF LEN(orders_to_reassign) > 0 THEN:
            available = FILTER(drones, is_available AND id != event.drone_id)
            new_alloc = GreedyAllocate(orders_to_reassign, available, current_time)
            updated_allocation = MERGE(current_allocation, new_alloc)
            updated_allocation[low_batt_drone.id] = []
        ELSE:
            updated_allocation = current_allocation
        END IF

    // ----- 超时风险 -----
    TIMEOUT_RISK:
        updated_allocation = COPY(current_allocation)
        FOR each order IN orders WHERE status IN {ASSIGNED, PICKING}:
            assigned_drone = FIND_ASSIGNED_DRONE(order.id, drones, current_allocation)
            dist = DISTANCE(assigned_drone.position, order.delivery_pos)
            est_time = dist / assigned_drone.avg_speed + 30.0

            IF est_time > (order.deadline - current_time) THEN:
                // 寻找更近的可用无人机
                best_drone = NULL
                best_time = INF
                FOR each drone IN drones WHERE is_available AND id != assigned_drone.id:
                    d = DISTANCE(drone.position, order.pickup_pos)
                    t = d / drone.avg_speed + DISTANCE(order.pickup_pos, order.delivery_pos)
                        / drone.avg_speed + 30.0
                    IF t < best_time AND t < (order.deadline - current_time) THEN:
                        best_time = t
                        best_drone = drone
                    END IF
                END FOR

                IF best_drone != NULL THEN:
                    // 执行订单转移
                    REMOVE order.id FROM updated_allocation[assigned_drone.id]
                    APPEND order.id TO updated_allocation[best_drone.id]
                END IF
            END IF
        END FOR

    // ----- 无人机故障 -----
    UAV_FAULT:
        fault_drone = FIND(drones, id == event.drone_id)
        fault_drone.status = FAULT

        fault_orders = FILTER(
            orders WHERE id IN current_allocation[fault_drone.id]
                      AND status != COMPLETED
        )
        FOR each order IN fault_orders:
            order.status = PENDING
        END FOR

        available = FILTER(drones, is_available AND id != event.drone_id)
        pending = FILTER(orders, status == PENDING)
        IF LEN(available) > 0 AND LEN(pending) > 0 THEN:
            new_alloc = HungarianAllocate(pending, available, current_time)
            updated_allocation = MERGE(current_allocation, new_alloc)
            updated_allocation[fault_drone.id] = []
        ELSE:
            updated_allocation = current_allocation
        END IF

    // ----- 定期重优化 -----
    PERIODIC_REOPT:
        balance = ComputeLoadBalance(drones)
        IF balance < 0.6 THEN:
            pending = FILTER(orders, status == PENDING)
            idle = FILTER(drones, is_available)
            IF LEN(pending) > 0 AND LEN(idle) > 0 THEN:
                new_alloc = GreedyAllocate(pending, idle, current_time)
                updated_allocation = MERGE(current_allocation, new_alloc)
            ELSE:
                updated_allocation = current_allocation
            END IF
        ELSE:
            updated_allocation = current_allocation
        END IF
        // 更新所有订单优先级
        UpdateAllPriorities(orders, current_time)

    END CASE

    RETURN updated_allocation
END FUNCTION
```

## 四、PPO训练循环伪代码

```
================================================================================
算法: PPO主训练循环
================================================================================

PROCEDURE TrainPPO(env, config):
    """
    输入: 仿真环境 env, 超参数配置 config
    输出: 训练后的策略参数 θ*
    """

    // ==================== 初始化 ====================
    obs_dim = 10 + MAX_ORDERS + 1       // 编码后状态维度 (如31)
    action_dim = 3                       // 三维加速度

    // 创建网络
    actor = ActorNetwork(obs_dim, action_dim, hidden=[256, 128, 64])
    critic = CriticNetwork(obs_dim, hidden=[256, 128, 64])

    // 权重初始化 (正交初始化)
    actor.InitializeWeights(Orthogonal(gain=0.01))
    critic.InitializeWeights(Orthogonal(gain=1.0))
    actor.log_std_head.bias = -0.5       // 初始探索

    // 创建优化器
    actor_optimizer = Adam(actor.parameters, lr=LEARNING_RATE)
    critic_optimizer = Adam(critic.parameters, lr=LEARNING_RATE)
    scheduler = CosineAnnealingLR(actor_optimizer, T_max=NUM_EPOCHS, eta_min=LEARNING_RATE_MIN)

    // 创建经验缓冲区
    buffer = ReplayBuffer(capacity=BUFFER_CAPACITY)

    // 收敛监控
    convergence_monitor = ConvergenceMonitor(window_size=100)
    early_stopping = EarlyStopping(patience=50, min_delta=0.01)

    // ==================== 主训练循环 ====================
    FOR epoch = 1 TO NUM_EPOCHS:

        // ============ 阶段1: 经验收集 ============
        episode_rewards = []

        FOR episode = 1 TO episodes_per_epoch:
            raw_state = env.reset()
            encoded_state = StateEncode(raw_state)
            episode_reward = 0

            FOR t = 1 TO STEPS_PER_EPOCH / episodes_per_epoch:

                // --- Actor推理 ---
                action, log_prob = SelectAction(actor, encoded_state, deterministic=FALSE)

                // --- Critic评估 ---
                value = critic(encoded_state)

                // --- 动作施加与环境交互 ---
                next_raw_state, reward, done, info = env.step(action)
                next_encoded_state = StateEncode(next_raw_state)

                // --- 存储经验 ---
                buffer.store(encoded_state, action, reward, next_encoded_state,
                             done, log_prob, value)

                episode_reward += reward
                encoded_state = next_encoded_state

                IF done THEN:
                    BREAK
                END IF
            END FOR

            episode_rewards.APPEND(episode_reward)
            convergence_monitor.update(episode_reward, info["success"])
        END FOR

        // ============ 阶段2: GAE优势估计 ============
        // 从buffer提取经验批次
        batch = buffer.get_all_recent()
        states = batch.states
        actions = batch.actions
        rewards = batch.rewards
        values = batch.values
        dones = batch.dones
        old_log_probs = batch.log_probs

        // 计算最后状态的价值
        next_value = critic(StateEncode(env.current_state))

        // GAE递推计算 (从后向前)
        advantages = ZEROS(LEN(rewards))
        returns = ZEROS(LEN(rewards))
        last_gae = 0

        FOR t = LEN(rewards)-1 DOWNTO 0:
            IF t == LEN(rewards) - 1 THEN:
                next_val = next_value
            ELSE:
                next_val = values[t + 1]
            END IF

            // TD误差
            delta = rewards[t] + GAMMA * next_val * (1 - dones[t]) - values[t]

            // GAE优势
            advantages[t] = delta + GAMMA * GAE_LAMBDA * (1 - dones[t]) * last_gae
            last_gae = advantages[t]

            // 回报
            returns[t] = advantages[t] + values[t]
        END FOR

        // 归一化优势
        advantages = (advantages - MEAN(advantages)) / (STD(advantages) + 1e-8)

        // ============ 阶段3: 策略更新 (K轮) ============
        FOR k = 1 TO K_UPDATES:

            // 随机打乱数据
            indices = SHUFFLE(RANGE(LEN(states)))

            FOR start = 0 TO LEN(states)-1 STEP BATCH_SIZE:
                end = MIN(start + BATCH_SIZE, LEN(states))
                batch_idx = indices[start:end]

                s_batch = states[batch_idx]
                a_batch = actions[batch_idx]
                old_lp_batch = old_log_probs[batch_idx]
                adv_batch = advantages[batch_idx]
                ret_batch = returns[batch_idx]

                // ---- Actor更新 ----
                new_log_probs = actor.get_log_prob(s_batch, a_batch)
                entropy = actor.get_entropy(s_batch)

                // 概率比
                ratio = EXP(new_log_probs - old_lp_batch)

                // 裁剪目标
                surr1 = ratio * adv_batch
                surr2 = CLAMP(ratio, 1 - CLIP_EPSILON, 1 + CLIP_EPSILON) * adv_batch
                actor_loss = -MIN(surr1, surr2).mean()

                // 熵正则化
                entropy_loss = -entropy.mean()

                // ---- Critic更新 ----
                values_pred = critic(s_batch)
                value_loss = MSE(values_pred, ret_batch)

                // ---- 总损失与梯度更新 ----
                total_loss = actor_loss + VALUE_COEF * value_loss + ENTROPY_COEF * entropy_loss

                actor_optimizer.zero_grad()
                critic_optimizer.zero_grad()
                total_loss.backward()
                clip_grad_norm_(actor.parameters, MAX_GRAD_NORM)
                clip_grad_norm_(critic.parameters, MAX_GRAD_NORM)
                actor_optimizer.step()
                critic_optimizer.step()

                // ---- 监控指标 ----
                approx_kl = MEAN(old_lp_batch - new_log_probs)
                clip_fraction = MEAN(ABS(ratio - 1) > CLIP_EPSILON)
            END FOR
        END FOR

        // ============ 阶段4: 学习率更新 ============
        scheduler.step()

        // ============ 阶段5: 日志记录 ============
        IF epoch % 10 == 0 THEN:
            PRINT "Epoch", epoch
            PRINT "  平均回报:", MEAN(episode_rewards)
            PRINT "  Actor损失:", actor_loss
            PRINT "  价值损失:", value_loss
            PRINT "  策略熵:", entropy.mean()
            PRINT "  KL散度:", approx_kl
            PRINT "  裁剪比例:", clip_fraction
            PRINT "  学习率:", actor_optimizer.lr
        END IF

        // ============ 阶段6: 模型保存 ============
        IF epoch % 50 == 0 THEN:
            SAVE checkpoint: {actor, critic, actor_optimizer, critic_optimizer}
        END IF

        // ============ 阶段7: 早停检查 ============
        converged, metrics = convergence_monitor.check_convergence()
        IF early_stopping.check(metrics["mean_reward"]) THEN:
            PRINT "早停触发, 训练结束"
            BREAK
        END IF

    END FOR

    RETURN actor, critic
END PROCEDURE
```

### 5.1 CTDE集中式训练

```
================================================================================
算法: CTDE-PPO集中式训练
================================================================================

PROCEDURE TrainCTDE_PPO(env, config):
    """
    集中式训练: 共享Actor + 集中式Critic + 全局奖励
    输入: 多无人机仿真环境 env, 超参数配置 config
    输出: 训练后的共享Actor参数 θ*
    """

    // ==================== 初始化 ====================
    N = config.num_uavs                  // 无人机数量 (如3)
    obs_dim = 11                         // 局部观测维度
    action_dim = 3                       // 动作维度
    global_state_dim = N * obs_dim + 10  // 全局状态维度 (N*11 + 共享信息)

    // 共享Actor网络 (所有无人机共用一套参数)
    shared_actor = ActorNetwork(obs_dim, action_dim, hidden=[256, 128, 64])

    // 集中式Critic网络 (训练时使用全局状态)
    central_critic = CriticNetwork(global_state_dim, hidden=[256, 128, 64])

    // 优化器
    actor_optimizer = Adam(shared_actor.parameters, lr=LEARNING_RATE)
    critic_optimizer = Adam(central_critic.parameters, lr=LEARNING_RATE)
    scheduler = CosineAnnealingLR(actor_optimizer, T_max=NUM_EPOCHS, eta_min=LEARNING_RATE_MIN)

    // 经验缓冲区 (各无人机共享)
    buffer = RolloutBuffer(capacity=BUFFER_CAPACITY, obs_dim, action_dim, N)

    // 奖励计算器与课程学习管理器
    reward_calc = RewardCalculator(REWARD_CONFIG)
    curriculum = CurriculumManager(initial_stage=1)

    // ==================== 主训练循环 ====================
    FOR epoch = 1 TO NUM_EPOCHS:

        stage_config = curriculum.get_stage_config()

        // ============ 阶段1: 分散式经验收集 ============
        global_state = env.reset()
        episode_rewards = []

        FOR step = 1 TO STEPS_PER_EPOCH:

            // --- 获取各无人机局部观测 ---
            observations = []
            FOR i = 0 TO N-1:
                o_i = GetLocalObservation(global_state, i)  // 11维
                observations.APPEND(o_i)
            END FOR
            obs_matrix = STACK(observations)                 // (N, 11)

            // --- 各无人机独立选择动作 (共享Actor) ---
            actions = ZEROS(N, action_dim)
            log_probs = ZEROS(N)

            FOR i = 0 TO N-1:
                a_i, lp_i = SelectAction(shared_actor, observations[i], deterministic=FALSE)
                actions[i] = a_i
                log_probs[i] = lp_i
            END FOR

            // --- 集中式Critic评估全局状态价值 ---
            encoded_global = EncodeGlobalState(obs_matrix, shared_info)
            V_t = central_critic(TENSOR(encoded_global))

            // --- 环境执行联合动作 ---
            next_global_state, global_reward, done, info = env.step(actions)

            // --- 存储经验 (每个无人机都存储) ---
            FOR i = 0 TO N-1:
                buffer.store(
                    observation = observations[i],
                    action = actions[i],
                    reward = global_reward,             // 全局奖励
                    value = V_t,
                    log_prob = log_probs[i],
                    done = done,
                    global_state = encoded_global
                )
            END FOR

            global_state = next_global_state
            IF done THEN:
                episode_rewards.APPEND(info["episode_reward"])
                global_state = env.reset()
                reward_calc.reset_episode()
            END IF
        END FOR

        // ============ 阶段2: GAE计算 ============
        // 获取最后状态的价值
        last_global = EncodeGlobalState(
            GetLocalObservations(global_state), shared_info
        )
        next_value = central_critic(TENSOR(last_global))

        // GAE递推
        advantages, returns = ComputeGAE(
            buffer.rewards, buffer.values, buffer.dones,
            next_value, GAMMA, GAE_LAMBDA
        )
        advantages = NORMALIZE(advantages)

        // ============ 阶段3: 策略更新 (K轮) ============
        FOR k = 1 TO K_UPDATES:
            indices = SHUFFLE(RANGE(buffer.ptr))

            FOR start = 0 TO buffer.ptr-1 STEP BATCH_SIZE:
                end = MIN(start + BATCH_SIZE, buffer.ptr)
                batch_idx = indices[start:end]

                batch = {
                    obs:           buffer.observations[batch_idx],     // 局部观测
                    actions:       buffer.actions[batch_idx],
                    old_log_probs: buffer.log_probs[batch_idx],
                    advantages:    advantages[batch_idx],
                    returns:       returns[batch_idx],
                    global_states: buffer.global_states[batch_idx]     // 全局状态
                }

                // --- Actor更新 (使用局部观测) ---
                new_log_probs = shared_actor.get_log_prob(batch.obs, batch.actions)
                entropy = shared_actor.get_entropy(batch.obs)

                ratio = EXP(new_log_probs - batch.old_log_probs)
                surr1 = ratio * batch.advantages
                surr2 = CLAMP(ratio, 1-CLIP_EPSILON, 1+CLIP_EPSILON) * batch.advantages
                actor_loss = -MIN(surr1, surr2).mean()
                entropy_loss = -entropy.mean()

                // --- Critic更新 (使用全局状态) ---
                values_pred = central_critic(batch.global_states)
                value_loss = MSE(values_pred, batch.returns)

                // --- 总损失与梯度更新 ---
                total_loss = actor_loss + VALUE_COEF * value_loss + ENTROPY_COEF * entropy_loss

                actor_optimizer.zero_grad()
                critic_optimizer.zero_grad()
                total_loss.backward()
                clip_grad_norm_(shared_actor.parameters, MAX_GRAD_NORM)
                clip_grad_norm_(central_critic.parameters, MAX_GRAD_NORM)
                actor_optimizer.step()
                critic_optimizer.step()
            END FOR
        END FOR

        // ============ 阶段4: 学习率更新 ============
        scheduler.step()

        // ============ 阶段5: 课程学习检查 ============
        success = info.get("success", FALSE)
        collision = info.get("collision", FALSE)
        curriculum.update(success, step, collision)
        IF curriculum.should_advance() THEN:
            curriculum.advance_stage()
        END IF

        // ============ 阶段6: 日志记录 ============
        IF epoch % 10 == 0 THEN:
            PRINT "Epoch", epoch
            PRINT "  课程阶段:", curriculum.current_stage
            PRINT "  平均回报:", MEAN(episode_rewards)
            PRINT "  Actor损失:", actor_loss.item()
            PRINT "  价值损失:", value_loss.item()
            PRINT "  策略熵:", entropy.mean().item()
        END IF

        // ============ 阶段7: 模型保存 ============
        IF epoch % 50 == 0 THEN:
            SAVE checkpoint: {shared_actor, central_critic, actor_optimizer, critic_optimizer}
        END IF

    END FOR

    RETURN shared_actor
END PROCEDURE
```

```
PROCEDURE DecentralizedExecute(shared_actor_path, env, max_steps):
    """
    分散式执行: 各无人机独立决策, 无需全局信息
    输入: 训练好的共享Actor模型路径, 环境, 最大步数
    输出: 总奖励, 步数, 是否成功
    """

    // ==================== 初始化 ====================
    N = env.num_uavs

    // 加载训练好的共享Actor
    actor = LOAD_MODEL(shared_actor_path)
    actor.eval()                       // 切换到评估模式

    // 可选: 初始化通信模块
    comm_modules = []
    IF ENABLE_COMM THEN:
        FOR i = 0 TO N-1:
            comm_modules.APPEND(EventTriggeredComm(uav_id=i))
        END FOR
    END IF

    // ==================== Episode执行 ====================
    global_state = env.reset()
    total_reward = 0.0

    FOR step = 1 TO max_steps:

        // --- 获取各无人机局部观测 ---
        observations = []
        FOR i = 0 TO N-1:
            o_i = GetLocalObservation(global_state, i)  // 11维
            observations.APPEND(o_i)
        END FOR

        // --- 各无人机独立决策 (无通信) ---
        actions = ZEROS(N, 3)
        FOR i = 0 TO N-1:
            // 使用确定性模式 (评估时不加噪声)
            actions[i], _ = SelectAction(actor, observations[i], deterministic=TRUE)
        END FOR

        // --- 可选: 事件触发通信 ---
        IF ENABLE_COMM THEN:
            FOR i = 0 TO N-1:
                events = comm_modules[i].check_events(observations[i], neighbors, step * DT)
                IF LEN(events) > 0 THEN:
                    // 触发通信, 交换位置和速度信息
                    FOR each event IN events:
                        BroadcastMessage(comm_modules[i], event, observations[i])
                    END FOR
                END IF
            END FOR
        END IF

        // --- 环境执行联合动作 ---
        global_state, reward, done, info = env.step(actions)
        total_reward += reward

        IF done THEN:
            BREAK
        END IF
    END FOR

    RETURN total_reward, step, info.get("success", FALSE)
END PROCEDURE
```

```
PROCEDURE SystemMainLoop(env_config, algo_config):
    """
    系统主循环, 每 Δt = 0.1s 执行一次
    整合任务分配、路径规划、状态更新
    """

    // ==================== 系统初始化 ====================
    // 创建环境
    env = UAVDeliveryEnvironment(env_config)

    // 创建无人机集合
    drones = []
    FOR i = 0 TO N-1:
        drones.APPEND(CreateDrone(i, initial_position[i], initial_energy))
    END FOR

    // 创建订单优先级队列
    priority_queue = PriorityManager()

    // 创建任务分配器
    allocator = AdaptiveAllocator()

    // 创建动态重分配器
    reallocator = DynamicReallocator()

    // 创建奖励计算器
    reward_calc = RewardCalculator(REWARD_CONFIG)

    // 加载预训练PPO策略
    actor = LOAD_MODEL("trained_actor.pt")
    actor.eval()

    // 初始化分配结果
    current_allocation = {drone.id: [] FOR drone IN drones}

    // ==================== 主循环 ====================
    current_time = 0.0
    last_reopt_time = 0.0
    done = FALSE

    WHILE NOT done:

        // ============ 1. 状态更新 ============
        FOR each drone IN drones:
            drone.position = env.get_position(drone.id)
            drone.velocity = env.get_velocity(drone.id)
            drone.remaining_energy = env.get_energy(drone.id)
            drone.status = env.get_status(drone.id)
        END FOR

        // 更新订单状态
        FOR each order IN env.orders:
            order.status = env.get_order_status(order.id)
        END FOR

        // ============ 2. 事件检测 ============
        events = []

        // 新订单检测
        new_orders = env.get_new_orders()
        FOR each order IN new_orders:
            priority_queue.add_order(order, current_time)
            events.APPEND({type: NEW_ORDER, order: order})
        END FOR

        // 电量预警检测
        FOR each drone IN drones:
            IF drone.energy_ratio < 0.2 AND drone.status != CHARGING THEN:
                events.APPEND({type: BATTERY_LOW, drone_id: drone.id})
            END IF
        END FOR

        // 超时风险检测
        FOR each order IN env.orders WHERE status IN {ASSIGNED, PICKING}:
            assigned_drone = FIND_ASSIGNED_DRONE(order.id, drones, current_allocation)
            IF assigned_drone != NULL THEN:
                est_arrival = EstimateArrivalTime(assigned_drone, order, current_time)
                IF est_arrival > order.deadline THEN:
                    events.APPEND({type: TIMEOUT_RISK, order_id: order.id})
                END IF
            END IF
        END FOR

        // 定期重优化检测
        IF current_time - last_reopt_time >= T_REOPT THEN:
            events.APPEND({type: PERIODIC_REOPT})
            last_reopt_time = current_time
        END IF

        // ============ 3. 任务分配决策 ============
        FOR each event IN events:
            updated_allocation = reallocator.handle_event(
                event, drones, env.orders,
                current_allocation, current_time
            )
            current_allocation = updated_allocation
        END FOR

        // 补充分配: 将优先级队列中的订单分配给空闲无人机
        WHILE NOT priority_queue.is_empty:
            highest_order = priority_queue.pop_highest()
            IF highest_order == NULL THEN BREAK END IF

            idle_drones = FILTER(drones, is_available AND LEN(task_list) == 0)
            IF LEN(idle_drones) > 0 THEN:
                sub_result = GreedyAllocate([highest_order], idle_drones, current_time)
                current_allocation = MERGE(current_allocation, sub_result)
            ELSE:
                priority_queue.add_order(highest_order, current_time)  // 放回队列
                BREAK
            END IF
        END WHILE

        // ============ 4. 路径规划 (PPO策略) ============
        all_actions = ZEROS(N, 3)

        FOR i = 0 TO N-1:
            drone = drones[i]

            IF drone.status IN {IDLE, FLYING} AND LEN(drone.task_list) > 0 THEN:
                // 计算局部观测
                obs = ComputeLocalObservation(drone, env)  // 11维
                encoded_obs = StateEncode(obs)

                // 使用PPO策略计算动作
                action, _ = SelectAction(actor, encoded_obs, deterministic=TRUE)
                all_actions[i] = action
            ELSE:
                // 空闲/充电/故障: 悬停或不动作
                all_actions[i] = [0, 0, 0]
            END IF
        END FOR

        // ============ 5. 动作执行与状态更新 ============
        prev_positions = COPY(positions)
        prev_actions = COPY(all_actions)

        global_state, reward, done, info = env.step(all_actions)

        // 更新无人机状态
        FOR i = 0 TO N-1:
            drones[i].position = env.get_position(i)
            drones[i].velocity = env.get_velocity(i)
            drones[i].remaining_energy = env.get_energy(i)
        END FOR

        // ============ 6. 奖励计算与记录 ============
        // (训练阶段)
        IF training_mode THEN:
            total_reward, components = reward_calc.calculate(
                positions = STACK([d.position FOR d IN drones]),
                prev_positions = prev_positions,
                actions = all_actions,
                prev_actions = prev_actions,
                velocities = STACK([d.velocity FOR d IN drones]),
                goal_pos = env.current_goal,
                goal_positions = env.delivery_goals,
                formation_offsets = GetFormationOffsets(N),
                obstacles = env.obstacles,
                cargo_loaded = env.cargo_loaded,
                cargo_delivered = env.cargo_delivered,
                battery_ratios = ARRAY([d.energy_ratio FOR d IN drones]),
                step_count = step,
                max_steps = MAX_STEPS
            )
            // 存储经验到buffer
            StoreExperience(buffer, observations, all_actions, total_reward, ...)
        END IF

        current_time += DT

        // ============ 7. 检查终止条件 ============
        IF ALL_ORDERS_COMPLETED(env.orders) THEN:
            done = TRUE
            PRINT "所有配送任务完成!"
        END IF
        IF current_time >= MAX_SIMULATION_TIME THEN:
            done = TRUE
            PRINT "仿真时间到达上限"
        END IF

    END WHILE

    // ==================== 输出结果 ====================
    results = {
        total_time: current_time,
        completed_orders: COUNT(env.orders, status == COMPLETED),
        total_distance: SUM([d.total_distance FOR d IN drones]),
        energy_consumed: SUM([d.initial_energy - d.remaining_energy FOR d IN drones]),
        load_balance: ComputeLoadBalance(drones)
    }

    RETURN results
END PROCEDURE
```

### 7.1 GAE优势估计

```
FUNCTION ComputeGAE(rewards, values, dones, next_value, gamma, gae_lambda) -> advantages, returns:
    """
    通用GAE优势估计 (从后向前递推)
    """

    T = LEN(rewards)
    advantages = ZEROS(T)
    returns = ZEROS(T)
    last_gae = 0

    FOR t = T-1 DOWNTO 0:
        IF t == T - 1 THEN:
            next_val = next_value
        ELSE:
            next_val = values[t + 1]
        END IF

        // TD误差
        delta = rewards[t] + gamma * next_val * (1 - dones[t]) - values[t]

        // GAE优势
        advantages[t] = delta + gamma * gae_lambda * (1 - dones[t]) * last_gae
        last_gae = advantages[t]

        // 回报
        returns[t] = advantages[t] + values[t]
    END FOR

    RETURN advantages, returns
END FUNCTION
```

### 7.2 局部观测计算

```
FUNCTION ComputeLocalObservation(drone, env) -> observation:
    """
    计算单架无人机的11维局部观测
    """

    observation = ZEROS(11)

    // 位置 (3维)
    observation[0] = drone.position.x
    observation[1] = drone.position.y
    observation[2] = drone.position.z

    // 速度 (3维)
    observation[3] = drone.velocity.x
    observation[4] = drone.velocity.y
    observation[5] = drone.velocity.z

    // 目标距离 (1维)
    goal_pos = env.get_current_goal(drone.id)
    observation[6] = DISTANCE(drone.position, goal_pos)

    // 障碍物距离 (1维)
    min_obs_dist = INF
    FOR each obs IN env.obstacles:
        d = DISTANCE(drone.position, obs) - obs.radius
        min_obs_dist = MIN(min_obs_dist, d)
    END FOR
    observation[7] = MAX(0, min_obs_dist)

    // 无人机间距离 (1维)
    min_uav_dist = INF
    FOR each other IN env.drones WHERE other.id != drone.id:
        d = DISTANCE(drone.position, other.position)
        min_uav_dist = MIN(min_uav_dist, d)
    END FOR
    observation[8] = min_uav_dist

    // 剩余电量 (1维)
    observation[9] = drone.remaining_energy / drone.max_energy

    // 配送进度 (1维)
    observation[10] = COUNT(drone.task_list, status == COMPLETED)

    RETURN observation
END FUNCTION
```

### 7.3 负载均衡计算

```
FUNCTION ComputeLoadBalance(drones) -> balance:
    """
    计算负载均衡度 Balance ∈ [0, 1]
    """

    load_metrics = []
    FOR each drone IN drones:
        task_factor = LEN(drone.task_list) / 10.0
        energy_factor = (drone.max_energy - drone.remaining_energy) / drone.max_energy
        load_metrics.APPEND(task_factor + energy_factor)
    END FOR

    IF ALL(load_metrics == 0) THEN:
        RETURN 1.0
    END IF

    mean_val = MEAN(load_metrics)
    std_val = STD(load_metrics)

    IF mean_val == 0 THEN:
        RETURN 1.0
    END IF

    balance = 1.0 - (std_val / mean_val)
    balance = CLIP(balance, 0.0, 1.0)

    RETURN balance
END FUNCTION
```

```
FUNCTION ComputePriority(order, current_time) -> priority_score:
    """
    计算订单综合优先级
    priority_total = priority_base + lambda_1 * urgency + lambda_2 * wait
    """

    base_priority = order.priority       // 1-5

    // 紧急度
    time_ratio = (order.deadline - current_time) / MAX(order.deadline - order.created_time, 1)
    IF time_ratio <= 0.2 THEN:
        urgency = 5
    ELSE IF time_ratio <= 0.5 THEN:
        urgency = 2
    ELSE:
        urgency = 0
    END IF

    // 等待时间
    wait_time = MIN(3, FLOOR((current_time - order.created_time) / 300))

    priority_score = base_priority + 0.6 * urgency + 0.4 * wait_time

    RETURN priority_score
END FUNCTION
```
